cmake_minimum_required(VERSION 3.10)

# Включаем поддержку языков: C и Ассемблер (ASM)
# Если ты используешь NASM, замени ASM на ASM_NASM
project(AFF_Archiver VERSION 1.0.0 LANGUAGES C ASM)

# Устанавливаем стандарт C99 или C11 для чистого и строгого кода
set(CMAKE_C_STANDARD 99)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)

# Настройки компилятора C: включаем все предупреждения для безопасной работы с памятью
if(MSVC)
    add_compile_options(/W4)
else()
    add_compile_options(-Wall -Wextra -O2)
endif()

# Указываем CMake, где искать заголовочные файлы (инclude-директории)
include_directories(${CMAKE_CURRENT_SOURCE_DIR}/src)

# Формируем список исходных файлов проекта
# Сюда входят твои Си-файлы и Ассемблерные исходники (.s или .asm)
set(SOURCE_FILES
    src/rle.c
    src/rle.h
    src/main.c
    # Раскомментируй строку ниже и укажи имя своего ассемблерного файла, когда добавишь его:
    # src/crypto.s 
)

# Создаем исполняемый файл архиватора aff
add_executable(aff ${SOURCE_FILES})

# Выводим красивое сообщение в консоль при генерации проекта
message(STATUS "AFF Archiver configured successfully with C and ASM support!")
